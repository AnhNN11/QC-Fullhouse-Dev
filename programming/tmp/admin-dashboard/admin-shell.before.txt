"use client";

import { useMemo, useState } from "react";
import { BookOpen, CheckCircle2, Code2, Database, ExternalLink, GraduationCap, LayoutDashboard, Loader2, LogOut, Pencil, Plus, Search, Send, Trash2, Users } from "lucide-react";
import type { AdminData } from "@/lib/db";
import type { Course, Problem } from "@/lib/types";
import { logoutAdmin } from "./actions";
import Brand from "../brand";
import { Badge } from "@/components/ui/badge";
import { Button } from "@/components/ui/button";
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { Textarea } from "@/components/ui/textarea";

type Editor = { type: "course"; item?: Course } | { type: "problem"; item?: Problem } | { type: "learner"; item: AdminData["learners"][number] } | null;

const statCards = [
  ["courses", "Khóa học", BookOpen, "Nội dung đang xuất bản"],
  ["problems", "Bài tập", Code2, "Trong kho luyện tập"],
  ["learners", "Học viên", Users, "Tài khoản hoạt động"],
  ["submissions", "Bài nộp", Send, "100 bài gần nhất"],
] as const;

export default function AdminShell({ initialData }: { initialData: AdminData }) {
  const [data, setData] = useState(initialData);
  const [editor, setEditor] = useState<Editor>(null);
  const [query, setQuery] = useState("");
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  const filteredCourses = useMemo(() => data.courses.filter((item) => `${item.title} ${item.level}`.toLowerCase().includes(query.toLowerCase())), [data.courses, query]);
  const filteredProblems = useMemo(() => data.problems.filter((item) => `${item.title} ${item.topic}`.toLowerCase().includes(query.toLowerCase())), [data.problems, query]);

  async function mutate(action: string, payload: Record<string, unknown>) {
    setSaving(true);
    setError("");
    try {
      const response = await fetch("/api/admin", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action, payload }) });
      const result = await response.json() as AdminData & { error?: string };
      if (!response.ok) throw new Error(result.error || "Không thể cập nhật dữ liệu.");
      setData(result);
      setEditor(null);
    } catch (reason) {
      setError(reason instanceof Error ? reason.message : "Không thể cập nhật dữ liệu.");
    } finally {
      setSaving(false);
    }
  }

  function submitEditor(formData: FormData) {
    if (!editor) return;
    if (editor.type === "course") {
      void mutate("saveCourse", {
        id: editor.item?.id,
        title: formData.get("title"),
        description: formData.get("description"),
        icon: formData.get("icon"),
        color: formData.get("color"),
        level: formData.get("level"),
        lessons: Number(formData.get("lessons")),
        students: formData.get("students"),
        modules: String(formData.get("modules") ?? "").split("\n").map((value) => value.trim()).filter(Boolean),
      });
    } else if (editor.type === "problem") {
      void mutate("saveProblem", {
        id: editor.item?.id,
        title: formData.get("title"),
        slug: editor.item?.slug,
        topic: formData.get("topic"),
        difficulty: formData.get("difficulty"),
        acceptance: formData.get("acceptance"),
        statement: formData.get("statement"),
        exampleInput: formData.get("exampleInput"),
        exampleOutput: formData.get("exampleOutput"),
        explanation: formData.get("explanation"),
        constraints: String(formData.get("constraints") ?? "").split("\n").map((value) => value.trim()).filter(Boolean),
        starterCode: formData.get("starterCode"),
      });
    } else {
      void mutate("updateLearner", {
        id: editor.item.id,
        name: formData.get("name"),
        level: Number(formData.get("level")),
        xp: Number(formData.get("xp")),
      });
    }
  }

  async function remove(type: "Course" | "Problem", id: string | number, title: string) {
    if (!window.confirm(`Xóa “${title}”? Nội dung, quyền truy cập và dữ liệu tiến độ liên quan sẽ bị xóa. Không thể hoàn tác.`)) return;
    await mutate(`delete${type}`, { id });
  }

  return (
    <div className="admin-page">
      <aside className="admin-sidebar">
        <Brand />
        <div className="admin-sidebar-label">ADMIN CONSOLE</div>
        <nav>
          <a href="#overview" className="active"><LayoutDashboard /> Tổng quan</a>
          <a href="#content"><Database /> Nội dung</a>
          <a href="/admin/academy"><BookOpen /> Video & tư vấn 1:1</a>
          <a href="/admin/newsletter"><BookOpen /> Đăng ký nhận tin</a>
          <a href="/admin/enrollments"><GraduationCap /> Đăng ký khóa học</a>
          <a href="/admin/community"><Users /> Cộng đồng & kiểm duyệt</a>
          <a href="/admin/contests"><Code2 /> Cuộc thi & xếp hạng</a>
          <a href="/admin/blocks"><Code2 /> Khối lệnh & điểm thực hành</a>
          <a href="/admin/library"><BookOpen /> Tài nguyên & lộ trình</a>
          <a href="/admin/judge"><BookOpen /> Máy chấm & giáo án</a>
          <a href="#activity"><Send /> Hoạt động</a>
        </nav>
        <div className="admin-sidebar-footer">
          <a href="/dashboard" target="_blank"><ExternalLink /> Xem ứng dụng</a>
          <form action={logoutAdmin}><Button variant="ghost" type="submit"><LogOut /> Đăng xuất</Button></form>
        </div>
      </aside>

      <main className="admin-main">
        <header className="admin-topbar">
          <div><p>DolphinX Programming</p><h1>Trung tâm quản trị</h1></div>
          <div className="admin-avatar">AD</div>
        </header>

        <section id="overview" className="admin-section">
          <div className="admin-heading"><div><h2>Tổng quan hệ thống</h2><p>Dữ liệu trực tiếp từ MongoDB Atlas.</p></div><Badge variant="secondary"><span className="admin-live-dot" /> Live</Badge></div>
          <div className="admin-stat-grid">
            {statCards.map(([key, label, Icon, description]) => <Card key={key}><CardContent className="admin-stat"><div><p>{label}</p><strong>{data.stats[key]}</strong><span>{description}</span></div><span><Icon /></span></CardContent></Card>)}
          </div>
        </section>

        <section id="content" className="admin-section">
          <Tabs defaultValue="courses">
            <div className="admin-content-toolbar">
              <TabsList><TabsTrigger value="courses"><BookOpen /> Khóa học</TabsTrigger><TabsTrigger value="problems"><Code2 /> Bài tập</TabsTrigger><TabsTrigger value="learners"><GraduationCap /> Học viên</TabsTrigger></TabsList>
              <div className="admin-search"><Search /><Input value={query} onChange={(event) => setQuery(event.target.value)} placeholder="Tìm kiếm nội dung..." /></div>
            </div>

            <TabsContent value="courses">
              <Card><CardHeader className="admin-card-header"><div><CardTitle>Quản lý khóa học</CardTitle><CardDescription>Tạo, chỉnh sửa và theo dõi nội dung khóa học.</CardDescription></div><Button onClick={() => setEditor({ type: "course" })}><Plus /> Thêm khóa học</Button></CardHeader><CardContent className="p-0"><Table><TableHeader><TableRow><TableHead>Khóa học</TableHead><TableHead>Trình độ</TableHead><TableHead>Bài học</TableHead><TableHead>Tiến độ mẫu</TableHead><TableHead className="text-right">Thao tác</TableHead></TableRow></TableHeader><TableBody>{filteredCourses.map((course) => <TableRow key={course.id}><TableCell><div className="admin-entity"><span className={`admin-course-icon ${course.color}`}>{course.icon}</span><div><strong>{course.title}</strong><small>{course.description}</small></div></div></TableCell><TableCell><Badge variant="outline">{course.level}</Badge></TableCell><TableCell>{course.lessons}</TableCell><TableCell>{course.progress}%</TableCell><TableCell className="text-right"><Button size="icon-sm" variant="ghost" aria-label={`Sửa ${course.title}`} onClick={() => setEditor({ type: "course", item: course })}><Pencil /></Button><Button size="icon-sm" variant="ghost" aria-label={`Xóa ${course.title}`} onClick={() => remove("Course", course.id, course.title)}><Trash2 /></Button></TableCell></TableRow>)}</TableBody></Table></CardContent></Card>
            </TabsContent>

            <TabsContent value="problems">
              <Card><CardHeader className="admin-card-header"><div><CardTitle>Kho bài tập</CardTitle><CardDescription>Quản lý đề bài, độ khó và code khởi tạo.</CardDescription></div><Button onClick={() => setEditor({ type: "problem" })}><Plus /> Thêm bài tập</Button></CardHeader><CardContent className="p-0"><Table><TableHeader><TableRow><TableHead>#</TableHead><TableHead>Bài tập</TableHead><TableHead>Chủ đề</TableHead><TableHead>Độ khó</TableHead><TableHead>Tỉ lệ đúng</TableHead><TableHead className="text-right">Thao tác</TableHead></TableRow></TableHeader><TableBody>{filteredProblems.map((problem) => <TableRow key={problem.id}><TableCell>{problem.id}</TableCell><TableCell className="font-medium">{problem.title}</TableCell><TableCell>{problem.topic}</TableCell><TableCell><Badge variant={problem.difficulty === "Khó" ? "destructive" : "secondary"}>{problem.difficulty}</Badge></TableCell><TableCell>{problem.acceptance}</TableCell><TableCell className="text-right"><Button size="icon-sm" variant="ghost" aria-label={`Sửa ${problem.title}`} onClick={() => setEditor({ type: "problem", item: problem })}><Pencil /></Button><Button size="icon-sm" variant="ghost" aria-label={`Xóa ${problem.title}`} onClick={() => remove("Problem", problem.id, problem.title)}><Trash2 /></Button></TableCell></TableRow>)}</TableBody></Table></CardContent></Card>
            </TabsContent>

            <TabsContent value="learners">
              <Card><CardHeader><CardTitle>Học viên</CardTitle><CardDescription>Tài khoản đăng ký thật. Chuỗi học được tính từ lịch sử hoạt động.</CardDescription></CardHeader><CardContent className="p-0"><Table><TableHeader><TableRow><TableHead>Học viên</TableHead><TableHead>Cấp độ</TableHead><TableHead>XP</TableHead><TableHead>Chuỗi học</TableHead><TableHead className="text-right">Thao tác</TableHead></TableRow></TableHeader><TableBody>{data.learners.map((learner) => <TableRow key={learner.id}><TableCell><div className="admin-learner"><span>{learner.initials}</span><div><strong>{learner.name}</strong><small className="block text-muted-foreground">{learner.email}</small></div></div></TableCell><TableCell>Lv. {learner.level}</TableCell><TableCell>{learner.xp.toLocaleString("vi-VN")}</TableCell><TableCell>{learner.streak} ngày</TableCell><TableCell className="text-right"><Button size="sm" variant="outline" onClick={() => setEditor({ type: "learner", item: learner })}><Pencil /> Chỉnh sửa</Button></TableCell></TableRow>)}</TableBody></Table></CardContent></Card>
            </TabsContent>
          </Tabs>
        </section>

        <section id="activity" className="admin-section">
          <Card><CardHeader><CardTitle>Bài nộp gần đây</CardTitle><CardDescription>Lịch sử chấm bài được ghi nhận từ phòng luyện code.</CardDescription></CardHeader><CardContent className="p-0"><Table><TableHeader><TableRow><TableHead>Học viên</TableHead><TableHead>Bài tập</TableHead><TableHead>Ngôn ngữ</TableHead><TableHead>Kết quả</TableHead><TableHead>Thời gian</TableHead><TableHead>Ngày nộp</TableHead></TableRow></TableHeader><TableBody>{data.submissions.length ? data.submissions.map((submission) => <TableRow key={submission.id}><TableCell>{submission.learnerName}</TableCell><TableCell className="font-medium">{submission.problemTitle}</TableCell><TableCell>{submission.language}</TableCell><TableCell>{submission.accepted ? <Badge className="admin-accepted"><CheckCircle2 /> Đã nhận</Badge> : <Badge variant="destructive">Chưa đạt</Badge>}</TableCell><TableCell>{submission.runtimeMs} ms</TableCell><TableCell>{new Intl.DateTimeFormat("vi-VN", { dateStyle: "short", timeStyle: "short" }).format(new Date(submission.createdAt))}</TableCell></TableRow>) : <TableRow><TableCell colSpan={6} className="h-28 text-center text-muted-foreground">Chưa có bài nộp nào.</TableCell></TableRow>}</TableBody></Table></CardContent></Card>
        </section>
      </main>

      <Dialog open={Boolean(editor)} onOpenChange={(open) => !open && setEditor(null)}>
        <DialogContent className="admin-dialog">
          <DialogHeader><DialogTitle>{editor?.type === "course" ? `${editor.item ? "Chỉnh sửa" : "Thêm"} khóa học` : editor?.type === "problem" ? `${editor.item ? "Chỉnh sửa" : "Thêm"} bài tập` : "Chỉnh sửa học viên"}</DialogTitle><DialogDescription>Thông tin được lưu trực tiếp vào MongoDB và cập nhật ngay trên ứng dụng.</DialogDescription></DialogHeader>
          {editor && <form action={submitEditor} className="admin-editor-form">
            {editor.type === "course" ? <CourseFields course={editor.item} /> : editor.type === "problem" ? <ProblemFields problem={editor.item} /> : <LearnerFields learner={editor.item} />}
            {error && <p className="admin-form-error" role="alert">{error}</p>}
            <DialogFooter><Button type="button" variant="outline" onClick={() => setEditor(null)}>Hủy</Button><Button type="submit" disabled={saving}>{saving && <Loader2 className="animate-spin" />}Lưu thay đổi</Button></DialogFooter>
          </form>}
        </DialogContent>
      </Dialog>
    </div>
  );
}

function Field({ label, name, defaultValue, type = "text", required = false }: { label: string; name: string; defaultValue?: string | number; type?: string; required?: boolean }) {
  return <div className="grid gap-2"><Label htmlFor={`admin-${name}`}>{label}</Label><Input id={`admin-${name}`} name={name} defaultValue={defaultValue} type={type} required={required} /></div>;
}

function CourseFields({ course }: { course?: Course }) {
  return <><div className="admin-form-grid"><Field label="Tên khóa học" name="title" defaultValue={course?.title} required /><Field label="Biểu tượng" name="icon" defaultValue={course?.icon ?? "</>"} /><Field label="Trình độ" name="level" defaultValue={course?.level ?? "Cơ bản"} /><Field label="Số bài học" name="lessons" type="number" defaultValue={course?.lessons ?? 1} /><Field label="Số học viên hiển thị" name="students" defaultValue={course?.students ?? "0"} /><div className="grid gap-2"><Label htmlFor="admin-color">Màu thẻ</Label><Select name="color" defaultValue={course?.color ?? "blue"}><SelectTrigger id="admin-color"><SelectValue /></SelectTrigger><SelectContent><SelectItem value="blue">Xanh dương</SelectItem><SelectItem value="violet">Tím</SelectItem><SelectItem value="yellow">Lam sáng</SelectItem><SelectItem value="coral">San hô</SelectItem><SelectItem value="green">Xanh lá</SelectItem><SelectItem value="navy">Navy</SelectItem></SelectContent></Select></div></div><div className="grid gap-2"><Label htmlFor="admin-description">Mô tả</Label><Textarea id="admin-description" name="description" defaultValue={course?.description} /></div><div className="grid gap-2"><Label htmlFor="admin-modules">Chương học (mỗi dòng một chương)</Label><Textarea id="admin-modules" name="modules" rows={5} defaultValue={course?.modules.join("\n")} /></div></>;
}

function ProblemFields({ problem }: { problem?: Problem }) {
  return <><div className="admin-form-grid"><Field label="Tên bài tập" name="title" defaultValue={problem?.title} required /><Field label="Chủ đề" name="topic" defaultValue={problem?.topic ?? "Mảng"} /><div className="grid gap-2"><Label htmlFor="admin-difficulty">Độ khó</Label><Select name="difficulty" defaultValue={problem?.difficulty ?? "Dễ"}><SelectTrigger id="admin-difficulty"><SelectValue /></SelectTrigger><SelectContent><SelectItem value="Dễ">Dễ</SelectItem><SelectItem value="Trung bình">Trung bình</SelectItem><SelectItem value="Khó">Khó</SelectItem></SelectContent></Select></div><Field label="Tỉ lệ đúng" name="acceptance" defaultValue={problem?.acceptance ?? "0%"} /></div><div className="grid gap-2"><Label htmlFor="admin-statement">Đề bài</Label><Textarea id="admin-statement" name="statement" rows={4} defaultValue={problem?.statement} /></div><div className="admin-form-grid"><Field label="Đầu vào mẫu" name="exampleInput" defaultValue={problem?.exampleInput} /><Field label="Đầu ra mẫu" name="exampleOutput" defaultValue={problem?.exampleOutput} /></div><Field label="Giải thích" name="explanation" defaultValue={problem?.explanation} /><div className="grid gap-2"><Label htmlFor="admin-constraints">Ràng buộc (mỗi dòng một mục)</Label><Textarea id="admin-constraints" name="constraints" defaultValue={problem?.constraints.join("\n")} /></div><div className="grid gap-2"><Label htmlFor="admin-starterCode">Code khởi tạo</Label><Textarea id="admin-starterCode" name="starterCode" className="admin-code-input" rows={8} defaultValue={problem?.starterCode ?? "function solve(input) {\n  return input;\n}"} /></div></>;
}

function LearnerFields({ learner }: { learner: AdminData["learners"][number] }) {
  return <div className="admin-form-grid"><Field label="Tên học viên" name="name" defaultValue={learner.name} required /><Field label="Cấp độ" name="level" type="number" defaultValue={learner.level} /><Field label="Điểm XP" name="xp" type="number" defaultValue={learner.xp} /><p>Chuỗi học: {learner.streak} ngày · Tự động tính từ hoạt động thực tế.</p></div>;
}
